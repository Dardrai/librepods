cmake_minimum_required(VERSION 3.22.1)
project(native_dongle_hid_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(dongle_hid_test dongle_hid_test.cpp)
target_include_directories(dongle_hid_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
    target_compile_options(dongle_hid_test PRIVATE /W4 /UNDEBUG)
else()
    target_compile_options(dongle_hid_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME dongle_hid_test COMMAND dongle_hid_test)
