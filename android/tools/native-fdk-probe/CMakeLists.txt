cmake_minimum_required(VERSION 3.22.1)
project(native_fdk_probe C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(NATIVE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../app/src/main/cpp" ABSOLUTE)
add_library(fdk_probe_backend STATIC "${NATIVE_ROOT}/eld/fdk_encoder.cpp")
target_include_directories(fdk_probe_backend PUBLIC "${NATIVE_ROOT}/eld")
add_library(fdk_probe_hash STATIC "${NATIVE_ROOT}/xz/xz_sha256.c")
target_include_directories(fdk_probe_hash PUBLIC "${NATIVE_ROOT}/xz")
if(ANDROID)
    add_executable(eld_fdk_probe probe.cpp)
    target_link_libraries(eld_fdk_probe fdk_probe_backend fdk_probe_hash dl m)
else()
    add_executable(fdk_encoder_test fdk_encoder_test.cpp)
    target_link_libraries(fdk_encoder_test fdk_probe_backend fdk_probe_hash)
    if(UNIX)
        target_link_libraries(fdk_probe_backend PUBLIC dl)
    endif()
endif()
