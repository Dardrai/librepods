cmake_minimum_required(VERSION 3.22.1)
project(hook_transaction_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
get_filename_component(ELD "${CMAKE_CURRENT_SOURCE_DIR}/../../app/src/main/cpp/eld" ABSOLUTE)
add_executable(hook_transaction_test hook_transaction_test.cpp "${ELD}/hook_transaction.cpp")
target_include_directories(hook_transaction_test PRIVATE "${ELD}")
target_compile_options(hook_transaction_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME hook_transaction_test COMMAND hook_transaction_test)
